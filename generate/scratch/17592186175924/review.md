\begin{aligned}y &= \frac{37}{3} -\frac{2 x}{3} \\ \frac{2 x}{3} + y &= \frac{37}{3} \\ 3(\frac{2 x}{3} + y) &= 3(\frac{37}{3}) \\ 2 x + 3 y &= 37\end{aligned}
$$
