\begin{aligned}\frac{\text{d}u}{\text{d}x} & ={2}^{x}\operatorname{ln}⁡2\,⟹\,\frac{1}{\operatorname{ln}⁡2}\,\text{d}u={2}^{x}\,\text{d}x.\end{aligned}

$$
\begin{aligned}\int {2}^{x+1}\sqrt{{2}^{x}-5}\,\text{d}x & =\int 2\cdot {2}^{x}\sqrt{{2}^{x}-5}\,\text{d}x \\  & =2\int \sqrt{{2}^{x}-5}\cdot {2}^{x}\,\text{d}x \\  & =2\int \sqrt{u}\cdot \frac{1}{\operatorname{ln}⁡2}\,\text{d}u \\  & =\frac{2}{\operatorname{ln}⁡2}\int {u}^{1/2}\,\text{d}u \\  & =\frac{2}{\operatorname{ln}⁡2}\cdot \frac{{u}^{3/2}}{(\frac{3}{2})}+C \\  & =\frac{2}{\operatorname{ln}⁡2}\cdot \frac{2}{3}\cdot {u}^{3/2} \\  & =\frac{4}{3\operatorname{ln}⁡2}\sqrt{{u}^{3}}+C \\  & =\frac{4}{3\operatorname{ln}⁡2}\sqrt{({2}^{x}-5)^{3}}+C.\end{aligned}
