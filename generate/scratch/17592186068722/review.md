$$
u={x}^{3}+4.
$$

$$
\begin{aligned}\frac{\text{d}u}{\text{d}x} & =3{x}^{2}\,⟹\,\text{d}u=3{x}^{2}\,\text{d}x,\end{aligned}

$$

$$
\int {u}^{5}\,\text{d}u=\frac{{u}^{6}}{6}+C

$$
\int 3{x}^{2}({x}^{3}+4)^{5}\,\text{d}x=\frac{({x}^{3}+4)^{6}}{6}+C.
$$
