\begin{aligned}{(\frac{∂Q}{∂x}-\frac{∂P}{∂y})}_{(1/2,0)} & =\frac{1}{(\frac{1}{2}+0)}-\frac{1}{2} \\  & =2-\frac{1}{2} \\  & =\frac{3}{2}.\end{aligned}
$$
