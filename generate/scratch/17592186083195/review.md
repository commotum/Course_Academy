\begin{aligned}{\int }_{2}^{\infty }\frac{1730}{{x}^{3}}\,\text{d}x & ={lim}_{a\to \infty }{\int }_{2}^{a}\frac{1730}{{x}^{3}}\,\text{d}x \\  & =1730{lim}_{a\to \infty }{\int }_{2}^{a}{x}^{-3}\,\text{d}x \\  & =1730{lim}_{a\to \infty }{[(-\frac{1}{2}{x}^{-2})]}_{2}^{a} \\  & =-865\cdot {lim}_{a\to \infty }{[\frac{1}{{x}^{2}}]}_{2}^{a} \\  & =-865\cdot {lim}_{a\to \infty }(\frac{1}{{a}^{2}}-\frac{1}{{2}^{2}}) \\  & =-865(0-\frac{1}{4}) \\  & =\frac{865}{4}.\end{aligned}
$$
