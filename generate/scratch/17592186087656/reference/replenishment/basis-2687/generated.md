$$
(x-a)^{2}+(y-b)^{2}={r}^{2}.

$$
\begin{aligned}(x-a)^{2}+(y-b)^{2} & ={r}^{2} \\ (x-3)^{2}+(y+\frac{3}{2})^{2} & =6 \\ (x-3)^{2}+(y-(- \frac{3}{2}))^{2} & =(\sqrt{6})^{2}.\end{aligned}
