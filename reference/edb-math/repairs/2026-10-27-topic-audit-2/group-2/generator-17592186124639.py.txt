def display_powers(e):
 # Combine only powers already sharing a base; never fold a numerical multiplier
 # into an exponent or rewrite a logarithm's base.
 if e.is_Add:return S.Add(*(display_powers(a) for a in e.args))
 if not e.is_Mul:return e
 groups={};others=[]
 for a in e.args:
  if a.func==S.exp:base,exponent=S.E,a.args[0]
  elif a.is_Pow and a.exp.free_symbols:base,exponent=a.base,a.exp
  else:others.append(a);continue
  groups.setdefault(base,[]).append(exponent)
 return S.Mul(*others,*(S.exp(sum(es)) if b==S.E else b**sum(es) for b,es in groups.items()))

def tex(e):
 return MF.normalize_tex(S.latex(display_powers(e), mul_symbol=None, inv_trig_style='full', ln_notation=True, order='lex'))

def exprs(name,p):
 """Return ordered (held coefficient,target factor) components and complete bank.
 Each wrong expression is an explicit observed operation pattern, not perturbation.
 """
 u=X if p['target']=='x' else Y; v=Y if u==X else X

 if name=='sin_poly':

  bank=[a*u**n*S.cos(v)+b*v,a*u**n*S.cos(v)+b*u,a*u**(n-1)*S.sin(v)+b*u,C,a*n*v**(n-1)*S.sin(v)+b*u]
 elif name=='cos_poly':
  parts=[(a*v,u**n),(b*v**m,S.cos(u))];C=a*n*v*u**(n-1)-b*v**m*S.sin(u)
  bank=[C,a*n*u**(n-1)-b*S.sin(u),a*u**n+b*m*v**(m-1)*S.sin(u),a*u**n-b*m*v**(m-1)*S.sin(u),a*n*v*u**(n-1)+b*v**m*S.sin(u)]
 elif name=='log':

  bank=[C,a*v*u+b*v**n/u,a*v+b*v**n*S.log(u),a*u+b*n*v**(n-1)*S.log(u),a*u+b*n*v**(n-1)/u]
 elif name=='reciprocal':

  bank=[-a/v**2+b*n/u**(n+1),a*u/v+b*n*v/u**(n+1),C,-a*u/v**2-b/u**n,a/v-b*v/u**n]
 elif name=='asin':
  parts=[(a*v,S.asin(u)),(-b*v**2,S.Integer(1))];C=a*v/S.sqrt(1-u**2)
  bank=[C,C-2*b*v,a*S.asin(u)-2*b*v,a/S.sqrt(1-u**2),a*v*S.asin(u)]
 elif name=='exp_sum':

 elif name=='inverse_power':

def signed_join(items):
 return ''.join(('+' if i and not z.startswith('-') else '')+z for i,z in enumerate(items))
def derivative(g,u):return r'\frac{∂}{∂'+str(u)+'}('+tex(g)+')'
def product(h,g):
 if h==1:return tex(g)
 if h==-1:return '-'+tex(g)
 return tex(h*g)
def row_data(name,p):
 u,v,parts,bank,C=exprs(name,p)
 # Preserve each source family’s original term order, sign, exponent splitting,
 # held-coefficient extraction and unevaluated numerical derivative products.
 terms=[product(h,g) for h,g in parts]
 if name=='exp_sum': terms[0]=tex(S.exp(p['a']*v+p['b']*u))
 if name=='tan_power': terms[1]='-'+tex(p['b']*S.Integer(p['base'])**(u+v))
 F=signed_join(terms)
 split=[];extract=[];evaluated=[]
 for i,(h,g) in enumerate(parts):
  sign='-' if h.could_extract_minus_sign() else ''
  ah=-h if sign else h
  # Source constant_sin differentiates the held variable itself, whereas
  # asin's entirely held term is factored into d/du(1).
  split.append(sign+derivative(ah*g,u))
  extract.append(sign+tex(ah)+r'\cdot '+derivative(g,u))
  dg=S.sec(u)**2 if name=='tan_power' and i==0 else S.diff(g,u)
  evaluated.append(sign+tex(ah)+r'\cdot '+('('+tex(dg)+')' if name in ('sin_poly','reciprocal','constant_sin','inverse_power') or dg.could_extract_minus_sign() else tex(dg)))
 if name=='exp_sum':
  split[0]=derivative(S.exp(p['a']*v)*S.exp(p['b']*u),u)
  # Keep explicitly separated exponential factors even though SymPy can combine.
  split[0]=r'\frac{∂}{∂'+str(u)+'}('+tex(S.exp(p['a']*v))+r'\cdot '+tex(S.exp(p['b']*u))+')'
 if name=='tan_power':
  split[1]='-'+r'\frac{∂}{∂'+str(u)+'}('+tex(p['b']*S.Integer(p['base'])**v)+r'\cdot '+tex(S.Integer(p['base'])**u)+')'
 if name=='asin':evaluated[1]='-0'
 return F,[r'\frac{∂}{∂'+str(u)+'}('+F+')',signed_join(split),signed_join(extract),signed_join(evaluated),tex(C)]

def raw_layout(name,p,F,rows):
 """Parameterized source prose/masks, no original-payload return branch.
 Shared prose is preserved and mathematical spelling is normalized;
 generated solution stages remain the same five operations.
 """

 elif sub:problem='Find $'+op+'$ for $f(x,y)='+F+'.$'
 else:problem='Find $'+op+'$ given that $f(x,y)='+F+'.$'
 intro=('The notation ${f}_{'+u+'}$ denotes the partial derivative of $f$ with respect to $'+u+'.$\n\n') if sub and name!='cos_poly' else ''
 intro+='We differentiate the given function with respect to $'+u+',$ treating all other variables as constants.\n\n'

 intro+='Additionally, we use the '+rule+' for differentiation, as follows:\n\n'

 return MF.normalize_prose(problem),MF.normalize_prose(solution)

def equivalent(a,b):
 d=S.expand(S.expand_power_exp(a-b))
 return d==0 or (S.trigsimp(d)==0 if d.has(S.tan,S.sec,S.csc) else S.cancel(d)==0)

def instance(name,p,direction='forward'):
 """Choose the derivative role without changing the supplied function."""
 if name=='hard_cross_log':
  u=Y if p['target']=='y' else X;v=X if u==Y else Y
  a,b,c=map(S.Integer,(p['a'],p['b'],p['c']))
  oldparts=[(a*v,S.log(u)),(b*S.log(v),u),(S.exp(c*v),S.exp(u)),(v,u**-1)]
  F=sum(h*g for h,g in oldparts)
 else:
  u,v,oldparts,_,_=exprs(name,p);F=sum(h*g for h,g in oldparts)
 target=u if direction=='forward' else v
 held=v if target==u else u
 parts=[]
 for h,g in oldparts:
  term=S.expand_power_exp(h*g)
  coefficient,active=term.as_independent(target,as_Add=False)
  parts.append((coefficient,active))
 return target,held,F,parts

def active_diff(g,u):
 # The authored path uses the taught tangent derivative, not a CAS identity.
 if g==S.tan(u):return S.sec(u)**2
 return S.diff(g,u)

def partial_path(F,u):
 values=[]
 for term in S.Add.make_args(F):
  h,g=S.expand_power_exp(term).as_independent(u,as_Add=False)
  values.append(h*active_diff(g,u))
 return sum(values)

def bank_for(parts,u,v,F,profile):
 ds=[h*active_diff(g,u) for h,g in parts];C=sum(ds)
 if profile=='wrong-variable':
  W=partial_path(F,v);Z=F
  errors=['correct','reversed signs in derivative','differentiates the other variable',
          'other variable and reversed signs','leaves function unchanged']
 elif profile=='held-factors':
  W=sum(active_diff(g,u) for h,g in parts)
  Z=sum(S.diff(h,v)*active_diff(g,u) for h,g in parts)
  errors=['correct','reversed signs in derivative','drops held coefficients',
          'drops held coefficients and reverses signs','differentiates held coefficients too']
 else:raise ValueError('unknown complete profile')
 bank=[C,-C,W,-W,Z]
 for i,e in enumerate(bank):
  if e.has(S.zoo,S.nan,S.oo):raise ValueError('undefined bank')
  if any(equivalent(e,z) for z in bank[:i]):raise ValueError('colliding authored bank')
 return C,bank,errors

def derivation(name,p,direction):
 u,v,F,parts=instance(name,p,direction)
 C=sum(h*active_diff(g,u) for h,g in parts)
 # Preserve each original summand, including the explicit exponential splitting.
 if name=='hard_cross_log':
  original_u=Y if p['target']=='y' else X;original_v=X if original_u==Y else Y
  original=[p['a']*original_v*S.log(original_u),p['b']*original_u*S.log(original_v),S.exp(p['c']*original_v+original_u),original_v/original_u]
 else:
  _,_,old,_,_=exprs(name,p);original=[display_powers(h*g) for h,g in old]
 function=signed_join([tex(t) for t in original])
 op='f_{'+str(u)+'}(x,y)'
 intro='Differentiate with respect to $'+str(u)+'$, treating $'+str(v)+'$ as constant. '
 if name in ('exp_sum','tan_power','hard_cross_log'):
  exponent=next((t for t in original if t.has(S.exp)),None)
  if exponent is not None:
   factors=next((h,g) for h,g in parts if (h*g).has(S.exp))
   intro+='First write $'+tex(exponent)+'='+tex(factors[0])+r'\cdot '+tex(factors[1])+'$ to expose the held exponential factor. '
  elif name=='tan_power':
   B=S.Integer(p['base']);orig_u=X if p['target']=='x' else Y;orig_v=Y if orig_u==X else X
   intro+='Use $'+tex(B**(orig_u+orig_v))+'='+tex(B**v)+r'\cdot '+tex(B**u)+'$. '
 if any(g.as_powers_dict().get(u,0)<0 for h,g in parts):
  intro+='Write reciprocal powers as negative powers, using $'+r'\frac{1}{'+str(u)+r'^n}='+str(u)+r'^{-n}$ for positive $n$. '
 if name=='hard_cross_log':intro+='The logarithm of the held variable is a constant multiplier. '
 solution=intro+'Apply the sum and constant multiple rules term by term.\n\n'
 # Each term gets its own short display: no horizontal clipping of long sums.
 for i,((h,g),term) in enumerate(zip(parts,original)):
  rows=[derivative(term,u)+' &='+tex(h)+r'\cdot '+derivative(g,u),
        '&='+tex(h)+r'\cdot \left('+tex(active_diff(g,u))+r'\right)',
        '&='+tex(h*active_diff(g,u))]
  solution+='$$\n'+r'\begin{aligned}'+(r' \\'+'\n').join(rows)+r'\end{aligned}'+'\n$$\n\n'
 if len(parts)>2:
  # Break the final sum across continuation rows, preserving every term.
  terms=[tex(h*active_diff(g,u)) for h,g in parts]
  first=signed_join(terms[:2]);rest=signed_join(terms[2:])
  solution+='Combining the terms,\n\n$$\n'+r'\begin{aligned}'+op+' &='+first+r' \\ &\quad '+('+' if not rest.startswith('-') else '')+rest+r'.\end{aligned}'+'\n$$'
 else:solution+='Therefore $'+op+'='+tex(C)+'$.'
 return function,MF.normalize_prose(solution),C

@MF.checked_math_content
def generate(level,seed=0,template=None,parameters=None,direction=None,profile=None,**unused):
 if level not in ('easy','moderate','hard'):raise ValueError('difficulty must be easy, moderate or hard')
 rng=random.Random(seed)
 names=[n for n,s in SPECS.items() if s[1]==level]
 name=template or rng.choice(names)
 if name not in names:raise ValueError('template difficulty mismatch')
 direction=direction or rng.choice(('forward','reverse'))
 profile=profile or rng.choice(ERROR_PROFILES)
 if direction not in ('forward','reverse'):raise ValueError('direction')
 for attempt in range(250):
  p=dict(parameters) if parameters is not None else normal_parameters(name,rng)
  if set(p)!=set(SPECS[name][2]) or p['target'] not in ('x','y'):raise ValueError('parameter roles')
  for k,z in p.items():
   if k=='target':continue
   if isinstance(z,bool) or not isinstance(z,int):raise ValueError('integer parameter required')
   if k in ('n','m','k','base'):
    if not 2<=z<=4:raise ValueError('power/base bounds')
   elif not 1<=abs(z)<=9:raise ValueError('coefficient bounds')
  u,v,F,parts=instance(name,p,direction)
  try:C,bank,errors=bank_for(parts,u,v,F,profile)
  except ValueError:
   if parameters is not None:raise
   continue
  break
 else:raise ValueError('cannot construct selected complete profile')
 function,solution,final=derivation(name,p,direction)
 assert equivalent(C,final)
 domain_text=served_domain(name,p)
 problem=MF.normalize_prose('Find $f_{'+str(u)+'}(x,y)$ for $f(x,y)='+function+'$'+(', where '+domain_text if domain_text else '')+'.')
 values=[tex(e) for e in bank]
 qid=ident([VERSION,level,problem,solution,sorted(values)])
 fid=ident([qid,'selection'])
 answers=[{':answer/id':ident([fid,z]),':answer/type':':answer.type/math',':answer/value':z} for z in values]
 correct=answers[0][':answer/id'];error_by_id={a[':answer/id']:e for a,e in zip(answers,errors)}
 rng.shuffle(answers)
 q={':question/id':qid,':question/difficulty':':question.difficulty/'+level,':question/problem':problem,':question/worked-solution':solution,':question/answer-fields':[{':answer-field/id':fid,':answer-field/key':'selection',':answer-field/type':':answer-field.type/radio',':answer-field/choices':answers,':answer-field/correct':[':answer/id',correct]}]}
 return {'knowledge_point':{'db_id':KP[':db/id'],'id':KP[':knowledge-point/id'],'title':KP[':knowledge-point/title']},'question':q,'generation':{'version':VERSION,'seed':seed,'template':name,'parameters':p,'direction':direction,'bank_profile':profile,'errors':error_by_id,'target_variable':str(u),'held_constant_variable':str(v),'source_id':SPECS[name][0],'source_basis':SOURCE['basis'],'displayed_final':tex(C),'eligible_for_learning':True,'runtime':{'python':sys.version.split()[0],'sympy':S.__version__}}}

def batch_cues(record):
 """Track incidental position and bank rank, never alter a mathematical key."""
 f=record['question'][':question/answer-fields'][0]
 correct=next(i for i,a in enumerate(f[':answer-field/choices'])
              if a[':answer/id']==f[':answer-field/correct'][1])
 # A bank-only rank at an ordinary in-domain point is an incidental cue.
 # Use the expressions already constructed, not an independent answer catalog.
 meta=record['generation']
 u,v,F,parts=instance(meta['template'],meta['parameters'],meta['direction'])
 _,bank,_=bank_for(parts,u,v,F,meta['bank_profile'])
 expressions={tex(b):b for b in bank}
 values=[float(expressions[a[':answer/value']].subs({X:S.Rational(2,5),Y:S.Rational(7,10)}))
         for a in f[':answer-field/choices']]
 rank=sorted(range(len(values)),key=lambda i:values[i]).index(correct)
 return correct,rank

def batch(level,count,seed,template=None):
 """Mix repeated families after profile rejection, preserving deterministic replay.

 A repeated visible family must not be answered by one position or numerical
 rank throughout the finite batch. Resample the same semantic slot, rather than
 removing its family, target direction or complete error profile.
 """
 if count<1:raise ValueError('positive count required')
 rng=random.Random(seed);names=[n for n,s in SPECS.items() if s[1]==level]
 if template:names=[template]
 slots=[(n,d,p) for n in names for d in ('forward','reverse') for p in ERROR_PROFILES]
 rng.shuffle(slots);seen=set();out=[];history={}
 for slot in range(count):
  n,d,p=slots[slot%len(slots)]
  for attempt in range(1000):
   r=generate(level,rng.getrandbits(63),template=n,direction=d,profile=p)
   qid=r['question'][':question/id']
   if qid in seen:continue
   cues=batch_cues(r)
   groups=(n,(n,d))
   if any(history.get(group) and any(
       len({old[i] for old in history[group]}|{cues[i]})==1 for i in range(2))
       for group in groups):continue
   seen.add(qid);out.append(r)
   for group in groups:history.setdefault(group,[]).append(cues)
   break
  else:raise ValueError('cannot construct a varied finite batch for '+n)
 return out
