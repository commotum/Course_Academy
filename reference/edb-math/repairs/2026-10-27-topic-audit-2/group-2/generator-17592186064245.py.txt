def tex(v):
 return s.latex(v).replace(r'\log',r'\ln').replace(r'\left','').replace(r'\right','')
def aligned(rows):return '$$\n'+r'\begin{aligned}'+r' \\ '.join(lhs+' & ='+rhs for lhs,rhs in rows)+r'\end{aligned}'+'\n$$'
def coeff(c,t):
 if c==1:return t
 if c==-1:return '-'+t
 return tex(c)+r'\cdot '+t

def mathdata(name,p):

 return u,z,rate,v,primitive

 if not 1<=p['a']<=8:raise ValueError('a must be 1..8')

def expanded_poly(expr):
 return s.expand(s.powsimp(expr,combine='exp',force=True),log=False,power_exp=False,power_base=False)

def poly(expr):

   if profile=='authored:differential':recipes=(recipes[0],'first_omitted_derivative_one')+recipes[2:]
   if profile=='authored:rate':recipes=recipes[:2]+('reciprocal_uv_derivative_one',)+recipes[3:]
  vals=[c]+[errors[e] for e in recipes]
  origin='authored complete operation bank'
 if any(v.has(s.zoo,s.nan,s.oo,s.I) for v in vals): raise ValueError('Invalid whole bank')

def authored_display(name,p,value):
 z=mathdata(name,p)[1];polynomial=expanded_poly(value/z)
 if TEMPLATES[name][3] in ('factor','base'):
  return r'\left('+tex(polynomial)+r'\right)'+tex(z)+'+C'
 terms=[]
 for degree,c in [(1,polynomial.coeff(X)),(0,polynomial.subs(X,0))]:
  if c==0:continue
  # Render each polynomial term times the unchanged exponential. Do not ask
  # expand() to split the exponential argument or expand logarithms of bases.
  term=tex(abs(c)*X**degree*z)
  prefix=(' - ' if c<0 else ' + ') if terms else ('-' if c<0 else '')
  terms.append(prefix+term)
 return (''.join(terms) or '0')+'+C'

# EASY: positive unshifted slope; one by-parts application.

def markdown(record):
 q=record['question'];f=q[':question/answer-fields'][0];g=record['generation']
 return '### '+q[':question/difficulty'].split('/')[-1]+' — '+g['template']+'\n\n'+q[':question/problem']+'\n\n'+'\n'.join('- $'+a[':answer/value']+'$'+(' **correct**' if a[':answer/id']==f[':answer-field/correct'][1] else '') for a in f[':answer-field/choices'])+'\n\n'+q[':question/worked-solution']

def batch(level,count,seed,template=None,profile=None):
 rng=random.Random(seed);names=[n for n,t in TEMPLATES.items() if t[0]==level]
 if template: names=[template]
 out=[];seen=set();schedule=[]
 for i in range(count):
  # Two prompt presentations share the same eight non-e base instances.
  # Once exhausted, continue the available exponential families rather than
  # failing a larger batch or recycling an already served semantic instance.
  active=[n for n in names if TEMPLATES[n][1]!='base' or len({r['generation']['parameters'].get('r') for r in out if 'r' in r['generation']['parameters']})<8]
  if not active:raise ValueError('Requested batch exceeds available semantic instances')
  schedule=[n for n in schedule if n in active]
  if not schedule: schedule=active[:];rng.shuffle(schedule)
  name=schedule.pop()
  for attempt in range(500):
   rec=generate(level,rng.getrandbits(64),name,profile=profile)
   key=rec['generation']['parameter_fingerprint']
   if key not in seen:break
  else: raise ValueError('Requested batch exceeds available distinct semantic instances')
  seen.add(key);out.append(rec)
 return out

def main():
 ap=argparse.ArgumentParser(description=__doc__);ap.add_argument('--difficulty',choices=['easy','moderate','hard','all'],default='moderate');ap.add_argument('--seed',type=int,default=0);ap.add_argument('--count',type=int,default=1);ap.add_argument('--template');ap.add_argument('--profile');ap.add_argument('--format',choices=['markdown','json'],default='json');ns=ap.parse_args()
 if ns.count<1:ap.error('Positive count required')
 records=[r for level in (['easy','moderate','hard'] if ns.difficulty=='all' else [ns.difficulty]) for r in batch(level,ns.count,ns.seed,ns.template,ns.profile)]
 print(json.dumps(records,indent=2,ensure_ascii=False) if ns.format=='json' else '\n\n---\n\n'.join(map(markdown,records)))
if __name__=='__main__':main()
