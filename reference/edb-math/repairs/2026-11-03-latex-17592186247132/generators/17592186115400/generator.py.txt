def frac(a,b): return r'\frac{'+a+'}{'+b+'}'
def par(a): return r'\left('+a+r'\right)'
def prod(a,b):
    if a=='1': return b
    if b=='1': return a
    return par(a)+r'\cdot '+par(b)
def mono(a,n):
    if n==0: return str(a)
    return ('' if a==1 else '-' if a==-1 else str(a)) + ('x' if n==1 else 'x^{'+str(n)+'}')
def add(a,b): return a + (b if b.startswith('-') else '+'+b)
def polynomial(a,n,b,m): return mono(a,n) if b==0 else add(mono(a,n),mono(b,m))
def inverse(name,u): return r'\operatorname{arc'+name+'}('+u+')'
def squared(u):
    if u=='x': return 'x^2'
    if u.startswith(r'\sqrt{') and u.endswith('}'): return u[6:-1]
    import re
    match=re.fullmatch(r'x\^\{([23])\}',u)
    if match: return 'x^{'+str(2*int(match[1]))+'}'
    return par(u)+'^2'
def inverse_denominator(name,u):
    return '1+'+(par(squared(u)) if u.startswith(r'\sqrt{') and squared(u)!='x' else squared(u)) if name=='cot' else '|'+u+'|'+r'\sqrt{'+squared(u)+'-1}'
def inverse_derivative(name,u,du):
    den=inverse_denominator(name,u)
    return ('' if name=='sec' else '-')+frac(du,den)
def compact_times(a,b):
    if a=='1': return b
    if b=='1': return a
    if a=='-1': return '-'+b
    if b=='-1': return '-'+a
    def wrap(t):
        # Additive polynomial factors need fences, a leading minus alone does not.
        return par(t) if '+' in t or '-' in t[1:] else t
    return wrap(a)+r'\,'+wrap(b)
def inner(power,shift=0,scale=1):
    if power=='sqrt':
        radical=r'\sqrt{'+polynomial(scale,1,shift,0)+'}'
        return radical,frac(str(scale),'2'+radical)
    power=int(power)
    return polynomial(scale,power,shift,0),mono(scale*power,power-1)
def elementary(kind,a=1,n=1,b=0,m=0,k=1,base=4):
    if kind=='poly': return polynomial(a,n,b,m),polynomial(a*n,n-1,b*m,m-1) if m else mono(a*n,n-1)
    if kind=='ln': return r'\ln(x)',frac('1','x')
    if kind=='exp':
        ex=mono(k,1); return 'e^{'+ex+'}', ('-' if k==-1 else str(k)+r'\cdot ' if k!=1 else '')+'e^{'+ex+'}'
    if kind=='sin': return r'\sin(x)',r'\cos(x)'
    if kind=='log': return r'\log_{'+str(base)+'}(x)',frac('1',r'x\ln('+str(base)+')')
    if kind=='sqrt': return r'\sqrt{x}',frac('1',r'2\sqrt{x}')
    raise ValueError(kind)
def uid(role,content): return str(uuid.uuid5(uuid.NAMESPACE_URL,'ca-inverse-products-v1:'+role+':'+content))

def make(level, *, kind='poly', inv='cot', power=1, shift=0,scale=1,a=1,n=1,b=0,m=0,k=1,base=4,reverse=False):
    params=dict(kind=kind,inv=inv,power=power,shift=shift,scale=scale,a=a,n=n,b=b,m=m,k=k,base=base)
    u,du=inner(power,shift,scale)
    if kind=='pair':
        p,dp=inverse('csc',u),inverse_derivative('csc',u,du)
        v,dv=inverse('sec',u),inverse_derivative('sec',u,du)
        domain=r'$|'+u+r'|>1$'
        formula = r"$\frac{d}{dx}"+p+"="+dp+r"$ and $\frac{d}{dx}"+v+"="+dv+"$."
    else:
        p,dp=(u,du) if kind=='sqrt' else elementary(kind,a,n,b,m,k,base)
        v,dv=inverse(inv,u),inverse_derivative(inv,u,du)
        domain = ('$'+squared(u)+'>0$' if kind=='sqrt' else '$x>0$' if kind in ('ln','log') else 'all real inputs')
        if inv!='cot': domain += r', with $|'+u+r'|>1$'
        formula = r"$\frac{d}{dx}"+v+'='+dv+'$.'
    function=prod(v,p) if reverse else prod(p,v)
    raw1,raw2=prod(dp,v),prod(p,dv)
    if kind=='pair':
        den=inverse_denominator('sec',u)
        term1='-'+frac(compact_times(du,v),den);term2=frac(compact_times(du,p),den);sign='+'
        # Fractions show each product contribution without a repeated wide factor.
        raw1,raw2=term1,term2
        numerator=p+'-'+v
        key=frac(numerator if du=='1' else compact_times(du,par(numerator)),den)
        conversions=['combine the two fractions over their common denominator']
    else:
        if kind in ('ln','log','sqrt'):
            den='x' if kind=='ln' else r'x\ln('+str(base)+')' if kind=='log' else '2'+u
            term1=frac(compact_times(str(scale),v) if kind=='sqrt' else v,den)
        else: term1=compact_times(dp,v)
        sign='+' if inv=='sec' else '-'
        term2=frac(compact_times(du,p),inverse_denominator(inv,u))
        conversions=['multiply the unchanged factor into the numerator']
        if kind=='sqrt':
            term2=frac(str(scale),'2(1+'+squared(u)+')')
            conversions.append('cancel the nonzero square root where its radicand is positive')
        key=term1+sign+term2
    final_rows = key if kind=='pair' else term1+r'\\'+'\n'+r'&\quad '+sign+term2
    problem="Calculate $f'(x)$ for $f(x)="+function+"$.\n\n{{derivative}}"
    # The learner's ordered derivation controls both key and final displayed line.
    intro='Work on the interior of the real domain: '+domain+'. '
    if kind=='pair': intro+='The two inverse derivative formulas and chain rule give '+formula+' Here the shared argument is $u(x)='+u+"$ with $u'(x)="+du+'$.'
    else:
        intro+='For the inverse factor, the derivative formula'
        if power!=1 or shift: intro+=' and chain rule'
        intro+=(' give ' if power!=1 or shift else ' gives ')+formula+' Here the argument and its derivative are $u(x)='+u+"$ and $u'(x)="+du+'$.'
    if power!=1 and kind!='pair':
        intro+=' The squared argument is $('+u+')^2='+squared(u)+'$.'
    compact_note='Move each unchanged factor into its fraction numerator.'
    if kind=='pair': compact_note='Combine the fractions over their common denominator.'
    if kind=='sqrt': compact_note+=' On the stated domain, cancel the nonzero factor $'+u+'$ and use $('+u+')^2='+squared(u)+'$.'
    solution=intro+'\n\nSet $p(x)='+p+"$ and $v(x)="+v+"$. Then $p'(x)="+dp+"$ and $v'(x)="+dv+'$. Apply the product rule:\n\n$$\n'+r'\begin{aligned}'+'\n'+r"f'(x)&=p'(x)v(x)+p(x)v'(x)"+r'\\'+'\n&='+raw1+r'\\'+'\n'+r'&\quad +'+raw2+'\n'+r'\end{aligned}'+'\n$$\n\n'+compact_note+'\n\n$$\n'+r'\begin{aligned}'+'\n'+r"f'(x)&="+final_rows+'\n'+r'\end{aligned}'+'\n$$'
    problem,solution,key=normalize_prose(problem),normalize_prose(solution),normalize_tex(key)
    identity=json.dumps([problem,solution,key,level],separators=(',',':'))
    aid=uid('answer',identity)

            ':answer-field/type':':answer-field.type/blank', ':answer-field/choices':[answer], ':answer-field/correct':[':answer/id',aid]}]}

         'inverse_factor':v,'inverse_derivative':dv,'terms':[term1,sign+term2],'final':key,'conversions':conversions}}}
    return validate_generated_record(r)

FAMILIES={'easy':['linear'], 'moderate':['polynomial','ln','exp','log'],
          'hard':['composite_ln','composite_sin','composite_exp','sqrt','pair']}

def specs(family):
    inverses=['cot','sec','csc']
    if family=='linear':
        return [dict(a=a,b=b,inv=v) for a,b,v in itertools.product([-3,-2,-1,1,2,3],range(-4,5),inverses)]
    if family=='polynomial':
        return [dict(a=a,n=n,b=b,m=m,inv=v) for a,n,b,m,v in itertools.product([-2,-1,1,2],[2,3],[-3,-2,-1,1,2,3],[0,1],inverses)]
    if family=='ln': return [dict(kind='ln',inv=v) for v in inverses]
    if family=='exp': return [dict(kind='exp',k=k,inv=v) for k,v in itertools.product([-2,-1,1,2,3],inverses)]
    if family=='log': return [dict(kind='log',base=b,inv=v) for b,v in itertools.product([2,3,4,5,7],inverses)]
    if family=='sqrt': return [dict(kind='sqrt',power='sqrt',inv='cot',scale=a,shift=b) for a,b in itertools.product([-2,-1,1,2],range(-2,3))]
    if family=='pair': return [dict(kind='pair',scale=a,shift=b) for a,b in itertools.product([-2,-1,1,2],range(-2,3))]
    kind=family.split('_')[1]
    return [dict(kind=kind,inv=v,power=p,shift=b) for v,p,b in itertools.product(inverses,[2,3],range(-2,3))]

def generate(level,count,seed):
    if count<0: raise ValueError('count must be nonnegative')
    rng=random.Random(seed); records=[]; seen=set()
    # Shuffle each semantic parameter deck, then interleave live families.
    # Exhausted finite families leave the schedule rather than forcing clones.
    decks={family:specs(family) for family in FAMILIES[level]}
    for deck in decks.values(): rng.shuffle(deck)
    while len(records)<count:
        schedule=[f for f,d in decks.items() if d]; rng.shuffle(schedule)
        if not schedule: raise ValueError('Requested count exhausts supported unique semantic functions')
        for family in schedule:
            if len(records)>=count: break
            r=make(level,**decks[family].pop())
            d=r['generation']['derivation']; signature=(d['factor'],d['inverse_factor'])
            if signature in seen: continue
            seen.add(signature); r['generation'].update(seed=seed,sample_index=len(records),sampling_family=family)
            records.append(r)
    return records
